💡 Fast Export Bookmarklet
To skip waiting for LinkedIn's email:
  1. Visit your LinkedIn Connections Page.
  2. Open Browser DevTools Console (F12 or right-click > Inspect > Console).
  3. Copy and paste this script, then press Enter: 
    c=[];(async()=>{for(i=0;i<5;i++){window.scrollTo(0,document.body.scrollHeight);await new Promise(r=>setTimeout(r,800))}document.querySelectorAll('.mn-connection-card').forEach(e=>{let n=e.querySelector('.mn-connection-card__name')?.innerText;let p=e.querySelector('.mn-connection-card__occupation')?.innerText;let co=p?.split(' at ')[1]||'';let po=p?.split(' at ')[0]||p;if(n&&co)c.push({f:n.split(' ')[0],l:n.split(' ').slice(1).join(' '),co:co,po:po})});let csv='First Name,Last Name,Company,Position\n'+c.map(r=>`"${r.f}","${r.l}","${r.co}","${r.po}"`).join('\n');let b=new Blob([csv],{type:'text/csv'});let a=document.createElement('a');a.href=URL.createObjectURL(b);a.download='Connections.csv';a.click()})()
  4. It instantly downloads Connections.csv!
